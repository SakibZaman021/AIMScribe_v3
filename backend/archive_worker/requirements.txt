# AIMScribe Archive Worker - AIMS LAB server
#
# Deliberately small. This process runs on the machine holding every consultation
# recording, so every dependency is attack surface. WAV joining, hashing and path
# handling are all standard library; what is here is an HTTP client, and - for the
# cloud copy - a FLAC codec and a cipher.

requests>=2.32.0

# Windows ships no system timezone database, and slim Linux images often omit it.
# Without this, zoneinfo cannot resolve Asia/Dhaka and file naming silently falls
# back to UTC - which files every evening consultation under the previous day.
tzdata>=2024.1

# The lossless cloud copy (SRS-ARC-11): FLAC encoding and decoding, through
# libsndfile. Only the worker needs it; the server never touches audio.
soundfile>=0.12.1

# AES-256-GCM for the copy, encrypted here before it leaves UIU (SRS-DAT-08).
cryptography>=42.0.0
